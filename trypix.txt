local function log(...)
    local t = {}
    for i = 1, select("#", ...) do t[#t + 1] = tostring(select(i, ...)) end
    local line = ("[%s] %s"):format(os.date("%H:%M:%S"), table.concat(t, " "))
    LOG[#LOG + 1] = line
    if #LOG > 120 then table.remove(LOG, 1) end
    print("[CTD]", line)
    if renderLog then task.spawn(renderLog) end
end

local function norm(s) return (tostring(s):lower():gsub("[%s%-_]+", " "):gsub("^%s+", ""):gsub("%s+$", "")) end
local function isGem(n) return tostring(n):lower():find("gem", 1, true) ~= nil end

local function fetch(url, hdr)
    local req = (syn and syn.request) or (http and http.request) or http_request or request
    if req then
        local ok, res = pcall(req, { Url = url, Method = "GET", Headers = hdr or {} })
        if ok and res then return res.Body or res.body end
    end
    local ok, b = pcall(function() return game:HttpGet(url, true) end)
    return ok and b or nil
end

local function parseItems(html)
    local b = html:gsub('\\"', '"'):gsub("\\\\/", "/")
    local out, pos = {}, 1
    while true do
        local s = b:find('{"id":"', pos, true)
        if not s then break end
        local nxt = b:find('{"id":"', s + 1, true) or (#b + 1)
        local c = b:sub(s, nxt)
        pos = nxt
        if c:find('"displayFields"', 1, true) then
            local name = c:match('"name":"(.-)"')
            local raw = c:match('"label":"Value","value":(%b{})') or c:match('"label":"Value","value":"(.-)"') or c:match('"label":"Value","value":([%d%.]+)')
            if name and raw then
                local v
                if raw:sub(1, 1) == "{" then
                    local lo, hi = raw:match('"low":(%d+),"high":(%d+)')
                    if lo then v = (tonumber(lo) + tonumber(hi)) / 2 end
                else v = tonumber(raw) end
                local k = norm(name)
                if k ~= "" and v then out[k] = { name = name, value = v } end
            end
        end
    end
    return out
end

local function parseMd(md)
    local out, cur = {}, nil
    for line in md:gmatch("[^\n]+") do
        local n = line:match("^###%s*%[([^%]]+)%]")
        if n then
            cur = n
        else
            local v = line:match("^Value%s*([%d,]+)")
            if cur and v then
                local k = norm(cur)
                local lo = tonumber((v:gsub(",", "")))
                if lo and k ~= "" then out[k] = { name = cur, value = lo } end
                cur = nil
            end
        end
    end
    return out
end

local function merge(a, b) for k, v in pairs(b) do a[k] = v end return a end

local function loadValues(force)
    if loading then return DBSTATE ~= "STALE" end
    if not force and os.clock() - lastLoad < CFG.refresh then return DBSTATE ~= "STALE" end
    loading = true
    local out, best = {}, 0
    for _ = 1, 3 do
        local body = fetch(CFG.valueUrl, HDR)
        if body and body:find('"items"', 1, true) then
            local p = parseItems(body)
            local n = 0
            for _ in pairs(p) do n = n + 1 end
            if n > best then best = n merge(out, p) end
            if n >= 40 then break end
        end
        task.wait(1)
    end
    if best < 40 then
        local md = fetch("https://r.jina.ai/" .. CFG.valueUrl)
        if md then
            local p = parseMd(md)
            local n = 0
            for _ in pairs(p) do n = n + 1 end
            if n > best then best = n merge(out, p) end
        end
    end
    if best > 0 then DB = merge(DB, out) end
    DBSTATE = best >= 40 and "LIVE" or (best > 0 and "PARTIAL" or "STALE")
    lastLoad = os.clock()
    loading = false
    local total = 0
    for _ in pairs(DB) do total = total + 1 end
    log("values", DBSTATE, best, "/", total)
    onChange:Fire()
    return DBSTATE ~= "STALE"
end

local function valueOf(n) local e = DB[norm(n)] return e and e.value or 0 end

local function inv()
    local l, p = {}, Profiles[LP.UserId]
    if p and type(p.Towers) == "table" then
        for uuid, t in pairs(p.Towers) do
            if type(t) == "table" and type(t.Name) == "string" and DB[norm(t.Name)] then
                UIndex[uuid] = t.Name
                l[#l + 1] = { uuid = uuid, name = t.Name, cat = t.ItemType or "Tower", value = valueOf(t.Name) }
            end
        end
    end
    table.sort(l, function(a, b) return a.value < b.value end)
    return l
end

local function score(c, gems)
    local t, items, unk = 0, {}, 0
    for k, n in pairs(c) do
        local e = DB[k]
        if e then
            t = t + e.value * n
            items[#items + 1] = { name = e.name, value = e.value, qty = n, gem = isGem(e.name) }
        else
            unk = unk + n
            items[#items + 1] = { name = k, value = 0, qty = n, unknown = true }
        end
    end
    table.sort(items, function(a, b) return a.value > b.value end)
    return t + (gems or 0) * CFG.gemValue, items, unk
end

local function evaluate()
    local tv, ti, unk = score(S.their, S.theirGems)
    local ov, oi = score(S.our, S.ourGems)
    local d = tv - ov
    local gem = S.theirGems > 0
    for _, it in ipairs(ti) do if it.gem then gem = true break end end
    return { their = tv, our = ov, delta = d, pct = ov > 0 and d / ov or (d > 0 and 999 or 0),
             tItems = ti, oItems = oi, gem = gem, tg = S.theirGems, og = S.ourGems, unknown = unk }
end

local function profitable(r)
    if DBSTATE == "STALE" then return false end
    if r.unknown > 0 then return false end
    if r.their < CFG.minTheirValue then return false end
    if CFG.requireGem and not r.gem then return false end
    if r.our == 0 and r.their > 0 then return true end
    return r.delta >= CFG.minWinAbs or r.pct >= CFG.minWinPct
end

local function resetTrade()
    S.tid, S.our, S.their, S.open, S.joined, S.slot = nil, {}, {}, false, false, 0
    S.ourGems, S.theirGems = 0, 0
end

local function addItem(uuid, cat)
    if not S.tid then return end
    S.slot = S.slot + 1
    if invoke(R.add, S.tid, CFG.slotStart + S.slot - 1, uuid, cat) then
        local nm = UIndex[uuid]
        if nm then S.our[norm(nm)] = (S.our[norm(nm)] or 0) + 1 end
        onChange:Fire()
    end
end

local function doAccept()
    if not S.tid or os.clock() - S.lastAccept < 3 then return end
    S.lastAccept = os.clock()
    local r = evaluate()
    SESSION.gain = SESSION.gain + r.delta
    SESSION.count = SESSION.count + 1
    if r.delta > SESSION.best then SESSION.best = r.delta end
    log("accept", ("%+d"):format(r.delta))
    fire(R.accept, S.tid, S.myOrd, "Accept")
    task.delay(CFG.confirmDelay, function() fire(R.accept, S.tid, S.myOrd, "Confirm") end)
    onChange:Fire()
end

local function doDecline()
    if not S.tid then return end
    fire(R.decline, S.tid, S.myOrd, "Decline")
    log("decline")
end

local function sideOf(pl)
    if type(pl) ~= "table" then return nil end
    if pl.Player == LP or pl.UserId == LP.UserId then return "me" end
    return "them"
end

local function countsOf(pl)
    local t = {}
    if type(pl) == "table" and type(pl.Items) == "table" then
        for _, it in pairs(pl.Items) do
            if type(it) == "table" and type(it.Name) == "string" then
                local k = norm(it.Name)
                t[k] = (t[k] or 0) + 1
            end
        end
    end
    return t
end

R.state.OnClientEvent:Connect(function(p)
    if type(p) ~= "table" then return end
    if type(p.TradeId) == "string" then S.tid = p.TradeId end
    local p1, p2 = p.Player1, p.Player2
    local mine, theirs
    if sideOf(p1) == "me" then mine, theirs = p1, p2
    elseif sideOf(p2) == "me" then mine, theirs = p2, p1
    else mine, theirs = p2, p1 end
    S.myOrd = (mine == p1) and 1 or 2
    S.our = countsOf(mine)
    S.their = countsOf(theirs)
    S.ourGems = type(mine) == "table" and tonumber(mine.CurrencyToGive) or 0
    S.theirGems = type(theirs) == "table" and tonumber(theirs.CurrencyToGive) or 0
    S.open = (p.State == "OPEN" and not p.Cancelled)
    if S.open and not S.joined and type(theirs) == "table" and theirs.Player then
        if CFG.autoJoin and (type(mine) ~= "table" or next(mine.Items or {}) == nil) then
            S.joined = true
            task.spawn(function() invoke(R.join, theirs.Player, S.tid) end)
        end
    end
    onChange:Fire()
end)

local blacklist, lastSend = {}, 0

local function targets()
    local out = {}
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= LP and not blacklist[p] then
            local dist = 9999
            local a = LP.Character and LP.Character:FindFirstChild("HumanoidRootPart")
            local b = p.Character and p.Character:FindFirstChild("HumanoidRootPart")
            if a and b then dist = (a.Position - b.Position).Magnitude end
            out[#out + 1] = { plr = p, dist = dist }
        end
    end
    table.sort(out, function(x, y) return x.dist < y.dist end)
    return out
end

local function offer()
    local cap = math.min(CFG.targetOffer, CFG.maxOfferValue)
    if CFG.offerGems > 0 and S.tid then invoke(R.currency, S.tid, CFG.offerGems) end
    local spent = 0
    for _, it in ipairs(inv()) do
        if spent >= cap then break end
        if not isGem(it.name) or spent < CFG.targetOffer * 0.6 then
            addItem(it.uuid, it.cat)
            spent = spent + it.value
            task.wait(0.25)
        end
    end
    return spent
end

local function watch()
    local t = os.clock()
    while os.clock() - t < CFG.tradeTimeout do
        task.wait(0.4)
        if not S.open then break end
        local r = evaluate()
        if profitable(r) then doAccept() return end
        if CFG.autoDecline and r.our > 0 and r.pct < -CFG.declineLossPct then doDecline() return end
    end
end

local function sendTrade(plr)
    if busy or not plr or not DBSTATE or DBSTATE == "STALE" then return end
    busy = true
    resetTrade()
    log("request", plr.Name)
    fire(R.send, plr)
    local t = os.clock()
    while os.clock() - t < CFG.openTimeout and not S.tid do task.wait(0.2) end
    if not S.tid then
        blacklist[plr] = true
        busy = false
        log("no window")
        return
    end
    S.joined = true
    if CFG.chatLine ~= "" then task.delay(1, function() fire(R.chat, S.tid, CFG.chatLine) end) end
    log("offered", offer())
    watch()
    task.wait(2)
    busy = false
end

local ADS = {}

local function loadAds()
    local body = fetch(CFG.adsUrl, HDR)
    if not body then log("ads fail") return end
    local t = body:gsub("<[^>]->", " "):gsub("&nbsp;", " "):gsub("&#x27;", "'"):gsub("%s+", " ")
    local out, pos = {}, 1
    while true do
        local s, e, user = t:find("Request Trade: message%s+([%w_]+)", pos)
        if not s then break end
        local nxt = t:find("Request Trade: message", e + 1) or (#t + 1)
        local seg = t:sub(e, nxt)
        pos = e + 1
        local function num(tag)
            local v, k = seg:match(tag .. "%s*([%d%.]+)%s*([Kk]?)")
            if not v then return nil end
            local n = tonumber(v)
            if not n then return nil end
            if k == "K" or k == "k" then n = n * 1000 end
            return n
        end
        local has, wants = num("Has"), num("Wants")
        if has and wants and has > wants then
            out[#out + 1] = { user = user, has = has, wants = wants, profit = has - wants }
        end
    end
    table.sort(out, function(a, b) return a.profit > b.profit end)
    ADS = out
    log("ads", #ADS)
    task.spawn(renderAds)
end

    b.TextSize = 11
    b.TextColor3 = Color3.fromRGB(185, 190, 205)
    b.Text = n
    b.Parent = remList
    b.MouseButton1Click:Connect(function()
        local o = EV:FindFirstChild(n) or RF:FindFirstChild(n)
        if not o then log(n, "missing") return end
        local a = argBox.Text
        local ok
        if o:IsA("RemoteFunction") then
            ok = invoke(o, a ~= "" and a or nil)
        else
            if a ~= "" then
                local plr = Players:FindFirstChild(a)
                fire(o, plr or a)
            else
                fire(o)
            end
            ok = true
        end
        log(n, ok and "fired" or "error")
    end)
end
