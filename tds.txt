local function norm(s) return (tostring(s):lower():gsub("[%s%-_]+", " "):gsub("^%s+", ""):gsub("%s+$", "")) end
local function isGem(n) return tostring(n):lower():find("gem", 1, true) ~= nil end

local function fetch(url, body, headers)
    local req = (syn and syn.request) or (http and http.request) or http_request or request
    if req then
        local ok, res = pcall(req, { Url = url, Method = body and "POST" or "GET", Body = body, Headers = headers or { ["User-Agent"] = "Mozilla/5.0" } })
        if ok and res then return res.Body or res.body end
    end
    if not body then local ok, b = pcall(function() return game:HttpGet(url, true) end) return ok and b or nil end
end

local function parseItems(html)
    local b = html:gsub('\\"', '"'):gsub("\\\\/", "/")
    local out, pos = {}, 1
    while true do
        local s = b:find('{"id":"', pos, true)
        if not s then break end
        local nxt = b:find('{"id":"', s + 1, true) or (#b + 1)
        local c = b:sub(s, nxt)
        pos = nxt
        if c:find('"displayFields"', 1, true) then
            local name = c:match('"name":"(.-)"')
            local raw = c:match('"label":"Value","value":(%b{})') or c:match('"label":"Value","value":"(.-)"') or c:match('"label":"Value","value":([%d%.]+)')
            if name and raw then
                local v
                if raw:sub(1, 1) == "{" then
                    local lo, hi = raw:match('"low":(%d+),"high":(%d+)')
                    if lo then v = (tonumber(lo) + tonumber(hi)) / 2 end
                else v = tonumber(raw) end
                local k = norm(name)
                if k ~= "" then out[k] = { name = name, value = v or CFG.ocValue, gem = isGem(name) } end
            end
        end
    end
    return out
end

local function loadValues()
    local b = fetch(CFG.valueUrl)
    if not b then log("fetch fail") return false end
    local p = parseItems(b)
    local n = 0
    for _ in pairs(p) do n = n + 1 end
    if n < 20 then log("parse fail " .. n) return false end
    DB = p
    log("values " .. n)
    return true
end

local function valueOf(name)
    local e = DB[norm(MAP[norm(name)] or name)]
    return e and e.value or 0, e
end

local function deepCollect(v, out, d)
    d = d or 0
    if d > 7 or type(v) ~= "table" then return out end
    local uuid, nm, cat
    for k, x in pairs(v) do
        if type(x) == "string" then
            if x:match("^%x%x%x%x%x%x%x%x%-%x%x%x%x%-") then uuid = x end
            local lk = tostring(k):lower()
            if lk == "name" or lk == "itemname" or lk == "item" then nm = x end
            if lk == "category" or lk == "type" or lk == "class" then cat = x end
        end
    end
    if uuid and nm then
        UUIDIndex[uuid] = nm
        if DB[norm(nm)] then out[#out + 1] = { uuid = uuid, name = nm, cat = cat or "Tower", value = valueOf(nm) } end
    end
    for _, x in pairs(v) do if type(x) == "table" then deepCollect(x, out, d + 1) end end
    return out
end

local function inventory()
    local l = {}
    for _, p in pairs(Profiles) do deepCollect(p, l) end
    table.sort(l, function(a, b) return a.value < b.value end)
    return l
end

local function groqMap()
    if not CFG.useGroq then return end
    local inv, seen, names = inventory(), {}, {}
    for _, it in ipairs(inv) do
        if not seen[it.name] then seen[it.name] = true names[#names + 1] = it.name end
    end
    if #names == 0 then log("groq: no inventory") return end
    local list = {}
    for _, e in pairs(DB) do list[#list + 1] = e.name end
    local prompt = "Match each inventory item to closest list item. JSON only {inv:list}.\nINV: " .. table.concat(names, " | ") .. "\nLIST: " .. table.concat(list, " | ")
    local body = HS:JSONEncode({ model = CFG.groqModel, temperature = 0, response_format = { type = "json_object" }, messages = { { role = "user", content = prompt } } })
    local res = fetch("https://api.groq.com/openai/v1/chat/completions", body, { ["Content-Type"] = "application/json", ["Authorization"] = "Bearer " .. CFG.groqKey })
    if not res then log("groq fail") return end
    local ok, data = pcall(HS.JSONDecode, HS, res)
    if not ok or not data or not data.choices then return end
    local ok2, m = pcall(HS.JSONDecode, HS, data.choices[1].message.content)
    if ok2 and type(m) == "table" then
        local n = 0
        for k, v in pairs(m) do if DB[norm(v)] then MAP[norm(k)] = v n = n + 1 end end
        log("mapped " .. n)
    end
end

local function bump(t, n, d)
    local k = norm(n)
    t[k] = math.max(0, (t[k] or 0) + d)
    if t[k] == 0 then t[k] = nil end
end

local function deepStrings(v, out, d)
    d = d or 0
    if d > 8 then return out end
    if type(v) == "string" then out[#out + 1] = v
    elseif type(v) == "table" then
        for k, x in pairs(v) do
            if type(k) == "string" then out[#out + 1] = k end
            deepStrings(x, out, d + 1)
        end
    end
    return out
end

local function score(counts)
    local t, items = 0, {}
    for k, n in pairs(counts) do
        local e = DB[k]
        if e then t = t + e.value * n items[#items + 1] = { name = e.name, value = e.value, qty = n, gem = e.gem } end
    end
    table.sort(items, function(a, b) return a.value > b.value end)
    return t, items
end

local function evaluate()
    local tv, ti = score(S.their)
    local ov, oi = score(S.our)
    local d = tv - ov
    local gem = false
    for _, it in ipairs(ti) do if it.gem then gem = true break end end
    return { their = tv, our = ov, delta = d, pct = ov > 0 and d / ov or (d > 0 and 999 or 0), tItems = ti, oItems = oi, gem = gem }
end

local function profitable(r)
    if r.their < CFG.minTheirValue then return false end
    if CFG.requireGem and not r.gem then return false end
    if r.our == 0 and r.their > 0 then return true end
    return r.delta >= CFG.minWinAbs or r.pct >= CFG.minWinPct
end

local function resetTrade()
    S.tradeId, S.our, S.their, S.open, S.slot = nil, {}, {}, false, 0
end

local function addItem(uuid, cat)
    if not S.tradeId then return end
    S.slot = S.slot + 1
    if invoke(R.add, S.tradeId, CFG.slotStart + S.slot - 1, uuid, cat) then
        local nm = UUIDIndex[uuid]
        if nm then bump(S.our, nm, 1) end
        onChange:Fire()
    end
end

local function removeItem(uuid, cat)
    if not S.tradeId then return end
    S.slot = math.max(0, S.slot - 1)
    invoke(R.remove, S.tradeId, CFG.slotStart + S.slot, uuid, cat)
    local nm = UUIDIndex[uuid]
    if nm then bump(S.our, nm, -1) end
    onChange:Fire()
end

local function doAccept()
    if not S.tradeId or os.clock() - S.lastAccept < 3 then return end
    S.lastAccept = os.clock()
    log("accept " .. S.tradeId)
    fire(R.accept, S.tradeId, CFG.acceptSlot, "Accept")
    task.delay(CFG.confirmDelay, function()
        fire(R.accept, S.tradeId, CFG.acceptSlot, "Confirm")
    end)
end

local function doDecline()
    if not S.tradeId then return end
    fire(R.decline, S.tradeId, CFG.acceptSlot, "Decline")
    task.delay(0.25, function() fire(R.decline, S.tradeId, CFG.acceptSlot) end)
    log("decline")
end

if R.state then
    R.state.OnClientEvent:Connect(function(...)
        local strs = deepStrings({ ... }, {})
        local flat = table.concat(strs, " "):lower()
        local tid = flat:match("(%x%x%x%x%x%x%x%x%-%x%x%x%x%-%x%x%x%x%-%x%x%x%x%-%x%x%x%x%x%x%x%x%x%x%x%x)")
        if tid then S.tradeId = tid end
        local claimed = {}
        for k, v in pairs(S.our) do claimed[k] = v end
        local theirs = {}
        for _, s in ipairs(strs) do
            local nm = DB[norm(s)] and s or UUIDIndex[s]
            if nm then
                local k = norm(nm)
                if DB[k] then
                    if (claimed[k] or 0) > 0 then claimed[k] = claimed[k] - 1
                    else theirs[k] = (theirs[k] or 0) + 1 end
                end
            end
        end
        S.their = theirs
        if flat:find("cancel", 1, true) or flat:find("declin", 1, true) then S.open = false
        elseif tid then S.open = true end
        onChange:Fire()
    end)
end

local busy, blacklist, lastSend = false, {}, 0

local function targets()
    local out = {}
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= LP and not blacklist[p] then
            local dist = 9999
            local a = LP.Character and LP.Character:FindFirstChild("HumanoidRootPart")
            local b = p.Character and p.Character:FindFirstChild("HumanoidRootPart")
            if a and b then dist = (a.Position - b.Position).Magnitude end
            out[#out + 1] = { plr = p, dist = dist }
        end
    end
    table.sort(out, function(x, y) return x.dist < y.dist end)
    return out
end

local function sendTrade(plr)
    if busy or not plr then return end
    busy = true
    resetTrade()
    log("request " .. plr.Name)
    fire(R.send, plr)
    local t = os.clock()
    while os.clock() - t < CFG.openTimeout and not (S.open and S.tradeId) do task.wait(0.2) end
    if not (S.open and S.tradeId) then
        blacklist[plr] = true
        busy = false
        log("no window")
        return
    end
    if CFG.chatLine ~= "" then task.delay(1, function() fire(R.chat, S.tradeId, CFG.chatLine) end) end
    local spent = 0
    local cap = math.min(CFG.targetOffer, CFG.maxOfferValue)
    for _, it in ipairs(inventory()) do
        if spent >= cap then break end
        if not it.gem or spent < CFG.targetOffer * 0.6 then
            addItem(it.uuid, it.cat)
            spent = spent + it.value
            task.wait(0.25)
        end
    end
    log("offered " .. spent)
    local t2 = os.clock()
    while os.clock() - t2 < CFG.tradeTimeout do
        task.wait(0.4)
        if not S.open then break end
        local r = evaluate()
        if profitable(r) then doAccept() break end
        if CFG.autoDecline and r.our > 0 and r.pct < -CFG.declineLossPct then doDecline() break end
    end
    task.wait(2)
    busy = false
end

task.spawn(function()
    while true do
        task.wait(1)
        if CFG.autoPipeline and not busy and os.clock() - lastSend > CFG.sendCooldown then
            local l = targets()
            if #l > 0 then lastSend = os.clock() sendTrade(l[1].plr) end
        end
    end
end)
